// CurrencyConverter.java

package currency;

import java.util.Scanner;

public class CurrencyConverter {
    Scanner sc = new Scanner(System.in);

    public void inrToEuro() {
        System.out.print("Enter amount in INR: ");
        double inr = sc.nextDouble();
        System.out.println("Euro : " + inr / 80);
    }

    public void euroToInr() {
        System.out.print("Enter amount in Euro: ");
        double euro = sc.nextDouble();
        System.out.println("Rupees : " + euro * 80);
    }

    public void inrToDollar() {
        System.out.print("Enter amount in INR: ");
        double inr = sc.nextDouble();
        System.out.println("Dollar : " + inr / 66);
    }

    public void dollarToInr() {
        System.out.print("Enter amount in Dollar: ");
        double dollar = sc.nextDouble();
        System.out.println("Rupees : " + dollar * 66);
    }

    public void inrToYen() {
        System.out.print("Enter amount in INR: ");
        double inr = sc.nextDouble();
        System.out.println("Yen : " + inr / 0.61);
    }

    public void yenToInr() {
        System.out.print("Enter amount in Yen: ");
        double yen = sc.nextDouble();
        System.out.println("Rupees : " + yen * 0.61);
    }
}


// DistanceConverter.java

package distance;

import java.util.Scanner;

public class DistanceConverter {
    Scanner sc = new Scanner(System.in);

    public void meterToKm() {
        System.out.print("Enter meters: ");
        double meter = sc.nextDouble();
        System.out.println("Kilometer : " + meter * 0.001);
    }

    public void kmToMeter() {
        System.out.print("Enter kilometers: ");
        double km = sc.nextDouble();
        System.out.println("Meter : " + km * 1000);
    }

    public void milesToKm() {
        System.out.print("Enter miles: ");
        double miles = sc.nextDouble();
        System.out.println("Kilometer : " + miles * 1.6093);
    }

    public void kmToMiles() {
        System.out.print("Enter kilometers: ");
        double km = sc.nextDouble();
        System.out.println("Miles : " + km / 1.6093);
    }
}


// TimeConverter.java

package time;

import java.util.Scanner;

public class TimeConverter {
    Scanner sc = new Scanner(System.in);

    public void hourToMinute() {
        System.out.print("Enter hours: ");
        double hour = sc.nextDouble();
        System.out.println("Minutes : " + hour * 60);
    }

    public void minuteToHour() {
        System.out.print("Enter minutes: ");
        double minute = sc.nextDouble();
        System.out.println("Hours : " + minute / 60);
    }

    public void hourToSecond() {
        System.out.print("Enter hours: ");
        double hour = sc.nextDouble();
        System.out.println("Seconds : " + hour * 3600);
    }

    public void secondToHour() {
        System.out.print("Enter seconds: ");
        double second = sc.nextDouble();
        System.out.println("Hours : " + second / 3600);
    }
}


// Convertor.java

import currency.CurrencyConverter;
import distance.DistanceConverter;
import time.TimeConverter;
import java.util.Scanner;

public class Convertor {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        CurrencyConverter currency = new CurrencyConverter();
        DistanceConverter distance = new DistanceConverter();
        TimeConverter time = new TimeConverter();

        System.out.println("1. Currency");
        System.out.println("2. Distance");
        System.out.println("3. Time");
        System.out.print("Enter code: ");
        int code = sc.nextInt();

        if (code == 1) {
            System.out.println("1. INR to Euro");
            System.out.println("2. Euro to INR");
            System.out.println("3. INR to Dollar");
            System.out.println("4. Dollar to INR");
            System.out.println("5. INR to Yen");
            System.out.println("6. Yen to INR");

            int ch = sc.nextInt();

            switch (ch) {
                case 1:
                    currency.inrToEuro();
                    break;
                case 2:
                    currency.euroToInr();
                    break;
                case 3:
                    currency.inrToDollar();
                    break;
                case 4:
                    currency.dollarToInr();
                    break;
                case 5:
                    currency.inrToYen();
                    break;
                case 6:
                    currency.yenToInr();
                    break;
                default:
                    System.out.println("Invalid Code");
            }
        }
        else if (code == 2) {
            System.out.println("1. Meter to KM");
            System.out.println("2. KM to Meter");
            System.out.println("3. Miles to KM");
            System.out.println("4. KM to Miles");

            int ch = sc.nextInt();

            switch (ch) {
                case 1:
                    distance.meterToKm();
                    break;
                case 2:
                    distance.kmToMeter();
                    break;
                case 3:
                    distance.milesToKm();
                    break;
                case 4:
                    distance.kmToMiles();
                    break;
                default:
                    System.out.println("Invalid Code");
            }
        }
        else if (code == 3) {
            System.out.println("1. Hour to Minute");
            System.out.println("2. Minute to Hour");
            System.out.println("3. Hour to Seconds");
            System.out.println("4. Seconds to Hour");

            int ch = sc.nextInt();

            switch (ch) {
                case 1:
                    time.hourToMinute();
                    break;
                case 2:
                    time.minuteToHour();
                    break;
                case 3:
                    time.hourToSecond();
                    break;
                case 4:
                    time.secondToHour();
                    break;
                default:
                    System.out.println("Invalid Code");
            }
        }
        else {
            System.out.println("Invalid Code");
        }
    }
}
